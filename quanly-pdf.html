function makeStampCanvas(text,size,mode){
  var scale=3,canvas=document.createElement('canvas'),ctx=canvas.getContext('2d');
  var fontSize=Math.max(8,Math.min(100,size))*scale;
  ctx.font='bold '+fontSize+'px Arial, "DejaVu Sans", sans-serif';
  var metrics=ctx.measureText(text),pad=mode==='watermark'?fontSize*.18:fontSize*.42;
  canvas.width=Math.ceil(metrics.width+pad*2);canvas.height=Math.ceil(fontSize*1.55+pad*2);
  ctx=canvas.getContext('2d');ctx.font='bold '+fontSize+'px Arial, "DejaVu Sans", sans-serif';
  ctx.textBaseline='middle';ctx.textAlign='center';
  if(mode==='stamp'){
    ctx.strokeStyle='#b91c1c';ctx.lineWidth=Math.max(2,scale*1.2);
    ctx.strokeRect(scale,scale,canvas.width-2*scale,canvas.height-2*scale);
  }
  ctx.fillStyle='#b91c1c';ctx.fillText(text,canvas.width/2,canvas.height/2);
  return canvas;
}
